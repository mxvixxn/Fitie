# Fitie v1.1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Close v1's core-loop gaps (day finalization, manual habits, weekday schedule UI) and add the retention layer (smart reminders, history calendar, per-habit completion insights) — all local, free-account, Simulator-verifiable.

**Architecture:** No schema changes. `RefreshService` generalizes to `refresh(day:dayIsOver:)`; finalization walks un-finalized past days (anchor in `UserDefaults`). Manual habits ride the existing `HabitEvaluator` (target 1 / measured 1) and are protected from auto overwrite via `ResultSource.manual`. Notifications move from repeating triggers to a 7-day rolling re-arm so achieved habits skip today. New UI (history calendar, completion card, weekday picker) follows the Apple-native design language in `Theme.swift` (`card()`, system colors).

**Tech Stack:** Swift 6, SwiftUI, SwiftData, HealthKit, UserNotifications, Swift Charts, Swift Testing, XcodeGen, xcodebuild + simctl.

**Spec:** `docs/superpowers/specs/2026-07-05-fitie-v1.1-design.md`

```swift
/// Evaluates every active habit for `day` and persists DailyResults.
/// `dayIsOver: true` finalizes the day (pending/inProgress collapse to missed).
func refresh(day: Date, dayIsOver: Bool, context: ModelContext) async throws {
    let day = Calendar.current.startOfDay(for: day)
    let habits = try context.fetch(FetchDescriptor<Habit>()).filter { !$0.isArchived }

    for habit in habits where habit.isActive(on: day) {
        let rule = habit.rule
        let existing = Store.dailyResult(context, habitID: habit.id, day: day)

        if rule.metric == .manual {
            // Manual habits are user-driven; refresh only closes out unchecked past days.
            if dayIsOver && existing?.status != .achieved {
                Store.upsertDailyResult(context, habitID: habit.id, day: day,
                                        status: .missed, measured: existing?.measured,
                                        source: existing?.source ?? .manual)
            }
            continue
        }
        // Never overwrite a user's manual override with auto data.
        if existing?.source == .manual { continue }

        let measured = (try? await health.value(for: rule.metric, on: day)) ?? nil
        let status = HabitEvaluator.status(measured: measured, target: rule.target,
                                           comparison: rule.comparison, dayIsOver: dayIsOver)
        Store.upsertDailyResult(context, habitID: habit.id, day: day,
                                status: status, measured: measured, source: .auto)
    }
    try? context.save()
}

/// Pulls today's health values while the day is still in progress.
func refreshToday(context: ModelContext, now: Date = Date()) async throws {
    try await refresh(day: now, dayIsOver: false, context: context)
}
```

```swift
/// Finalizes every day after `lastFinalizedDay` up to and including yesterday.
/// A nil anchor finalizes the trailing 7 days. Returns the new anchor
/// (yesterday), or nil when there was nothing to finalize.
func finalizePastDays(context: ModelContext, lastFinalizedDay: Date?,
                      now: Date = Date()) async throws -> Date? {
    let cal = Calendar.current
    let today = cal.startOfDay(for: now)
    let yesterday = cal.date(byAdding: .day, value: -1, to: today)!
    let anchor = lastFinalizedDay.map { cal.startOfDay(for: $0) }
        ?? cal.date(byAdding: .day, value: -7, to: today)!

    var day = cal.date(byAdding: .day, value: 1, to: anchor)!
    guard day <= yesterday else { return nil }
    while day <= yesterday {
        try await refresh(day: day, dayIsOver: true, context: context)
        day = cal.date(byAdding: .day, value: 1, to: day)!
    }
    return yesterday
}
```

import Foundation

enum ReminderPlanner {
    /// Concrete fire dates for the next `days` days at hour:minute.
    /// Today is dropped when its time already passed or `skipToday` is set
    /// (habit already achieved).
    static func fireDates(hour: Int, minute: Int, skipToday: Bool,
                          now: Date, days: Int = 7,
                          calendar: Calendar = .current) -> [Date] {
        var dates: [Date] = []
        for offset in 0..<days {
            guard let day = calendar.date(byAdding: .day, value: offset,
                                          to: calendar.startOfDay(for: now)),
                  let fire = calendar.date(bySettingHour: hour, minute: minute,
                                           second: 0, of: day) else { continue }
            if offset == 0 && (skipToday || fire <= now) { continue }
            dates.append(fire)
        }
        return dates
    }
}
```

/// Replaces this habit's pending reminders with the next 7 days of
/// one-shot notifications, skipping today when already achieved.
static func rearmReminder(habitID: UUID, name: String, hour: Int, minute: Int,
                          skipToday: Bool) async {
    await cancelReminder(habitID: habitID)
    let center = UNUserNotificationCenter.current()
    for fire in ReminderPlanner.fireDates(hour: hour, minute: minute,
                                          skipToday: skipToday, now: Date()) {
        let content = UNMutableNotificationContent()
        content.title = "Fitie"
        content.body = "\(name) 할 시간이에요."
        let comps = Calendar.current.dateComponents(
            [.year, .month, .day, .hour, .minute], from: fire)
        let request = UNNotificationRequest(
            identifier: "habit-\(habitID.uuidString)-\(Int(fire.timeIntervalSince1970))",
            content: content,
            trigger: UNCalendarNotificationTrigger(dateMatching: comps, repeats: false))
        try? await center.add(request)
    }
}

static func cancelReminder(habitID: UUID) async {
    let center = UNUserNotificationCenter.current()
    let ids = await center.pendingNotificationRequests()
        .map(\.identifier)
        .filter { $0.hasPrefix("habit-\(habitID.uuidString)") }
    center.removePendingNotificationRequests(withIdentifiers: ids)
}
```

import Foundation

enum CompletionStats {
    /// Achieved fraction (0...1) over the trailing window; nil when no
    /// decided (non-pending) results exist for the habit.
    static func rate(habitID: UUID, results: [DailyResult], windowDays: Int = 30,
                     now: Date = Date(), calendar: Calendar = .current) -> Double? {
        let cutoff = calendar.date(byAdding: .day, value: -windowDays,
                                   to: calendar.startOfDay(for: now))!
        let decided = results.filter {
            $0.habitID == habitID && $0.day >= cutoff && $0.status != .pending
        }
        guard !decided.isEmpty else { return nil }
        return Double(decided.filter { $0.status == .achieved }.count)
            / Double(decided.count)
    }
}

import Foundation
import SwiftData

enum DataExporter {
    struct Payload: Codable {
        struct HabitDTO: Codable {
            let id: UUID, name: String, colorHex: String
            let metric: String, comparison: String, target: Double
            let activeWeekdays: [Int], createdAt: Date, isArchived: Bool
        }
        struct ResultDTO: Codable {
            let habitID: UUID, day: Date, status: String
            let measured: Double?, source: String
        }
        struct ConditionDTO: Codable {
            let day: Date, mood: Int, energy: Int, note: String
        }
        let exportedAt: Date
        let habits: [HabitDTO]
        let results: [ResultDTO]
        let conditions: [ConditionDTO]
    }

    /// Serializes the whole store to a temp JSON file and returns its URL.
    @MainActor
    static func export(context: ModelContext) throws -> URL {
        let habits = try context.fetch(FetchDescriptor<Habit>())
        let results = try context.fetch(FetchDescriptor<DailyResult>())
        let conditions = try context.fetch(FetchDescriptor<ConditionEntry>())

        let payload = Payload(
            exportedAt: Date(),
            habits: habits.map {
                .init(id: $0.id, name: $0.name, colorHex: $0.colorHex,
                      metric: $0.rule.metric.rawValue,
                      comparison: $0.rule.comparison.rawValue,
                      target: $0.rule.target,
                      activeWeekdays: $0.activeWeekdays,
                      createdAt: $0.createdAt, isArchived: $0.isArchived)
            },
            results: results.map {
                .init(habitID: $0.habitID, day: $0.day, status: $0.statusRaw,
                      measured: $0.measured, source: $0.sourceRaw)
            },
            conditions: conditions.map {
                .init(day: $0.day, mood: $0.mood, energy: $0.energy, note: $0.note)
            })

        let encoder = JSONEncoder()
        encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
        encoder.dateEncodingStrategy = .iso8601
        let data = try encoder.encode(payload)

        let stamp = Date().formatted(.iso8601.year().month().day())
        let url = FileManager.default.temporaryDirectory
            .appendingPathComponent("Fitie-backup-\(stamp).json")
        try data.write(to: url)
        return url
    }
}
